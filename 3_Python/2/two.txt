  -> Reduce = it compress/reduce the list
     [LIB - functools]
     [SYN]
	reduce(<callback-function>, iterable[initializer])

     [USA]
	from functools import reduce
	numbers = [10, 20, 30, 40]
	res = reduce(lambda x, y:x if x>y else 	y, numbers)
	print(res)

  -> NamedTuple - creates object like the struct/pydantic basemodel
	[LIB] - collections
	[PAR]
		<tuple-name> = Gets tuple name
		<variable-name> = Gets tuple variables
	[SYN]
		<variable-name> = namedtuple(<tuple-name>, [variable-names])
	[USA]
		from collections import namedtuple
		Student = namedtuple("Student", ["name", "id", "marks"])
		s = Student("Steve Jobs", 111, 100)
		print(s.name)

  -> Set - Unique set of values
	[MET]
		remove(<value>) = remove if value available else throw error
		discard(<value>) = remove if value available else doesn't throw error
		pop() = remove random set value
		union() = combines sets [ALI] = '|'
		intersection() = get common values of sets [ALI] = '&'
		difference() = get set 1 values which is not in set 2 [ALI] = '-'
		symmetric_difference() = exclude common values [ALI] = '^'
	
	[CAT] - list inside set

  -> FrozenSet = make set immutable
	[SYN]
		<variable_name> = frozenset(<set_object>)
	[MET]
		=> union, intersection, difference, symmetric_difference

	Nested FrozenSet = combine frozenset in the separated nested format
	[RES]
		frozenset({frozenset(<set-1-values>), frozenset(<set-2-value>)})
	[USA]
		fs1 = {f1, f2}
	[SYN]
		<variable_name> = {<frozenset_object>, <frozenset_object>}
	
		[Note: works for frozenset, not for set]

  -> Combinations = returns all the combinations of the set as a individual tuple
	[SYN]
		combinations(start_index, end_index)
	[LIB]
		itertools
	[USA]
		from itertools import combinations
		power = {1, 2, 3}
		for r in range(len(power)):
			power.extend(combinations(power, r))

  	-> combine two set
	[RES]
		Error: unhashable
	    -> so combine two set values
	[SYN]
		<variable_name> = {*set_object, *set_object ...}
		[NOTE: asterisk unpack the values]
	[USA]
		a = {1, 2, 3}
		b = {3, 4}
		c = {*a, *b}
		print(a)
 
  -> Zip = combines the two set/list/tuple like a pair

	[SYN]

	[Note: if one object does not have pair it leave the data]
