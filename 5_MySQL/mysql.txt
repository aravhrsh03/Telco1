                "mysql+pymysql://root:root@localhost:3306", # verify port before use

            from sqlalchemy import create_engine, text, Column, Integer, String, text
            from sqlalchemy.orm import declarative_bass

            engine = create_engine(
                "mysql+pymysql://root:root@localhost:3306", # verify port before use
                isolation_level = "AUTOCOMMIT"
            )

            Base = declarative_bass()

            class emp1(Base):
                __tablename__ = "emp1"
                id = Column(Integer, primary_key = True, autoincrement = True)
                name = Column(String(100), nullable = False)
                phone = (Integer)
            
            Base.metadata.create_all(engine)
            print("table done")
        

    INSERT
        - To insert in SQLAlchemy we need session
        - we need to implement the class before insertion
        
            [USA]

                from sqlalchemy import create_engine, text, Column, Integer, String, text
                from sqlalchemy.orm import declarative_bass

                engine = create_engine(
                    "mysql+pymysql://root:root@localhost:3306", # verify port before use
                    isolation_level = "AUTOCOMMIT"
                )

                Base = declarative_bass()

                class emp1(Base):
                    __tablename__ = "emp1"
                    id = Column(Integer, primary_key = True, autoincrement = True)
                    name = Column(String(100), nullable = False)
                    phone = (Integer)
                
                Session = sessionmaker(bind = engine)
                session = Session()

                emp = emp1(name = "Madhan", phone = 1234567890)

                session.add(emp)
                session.commit()

                print("Record Inserted Successfully...")
                session.close()
