    [USA]
        from fastapi import FastAPI, HTTPException

        app = FastAPI()

        emp = []

        @app.get("/")
        def home():
            return {"message": "Hello"}

        # Get all employees
        @app.get("/emp")
        def get_data():
            return emp

        @app.post("/emp")
        def add_data(data: dict):
            emp.append(data)
            return {
                "message": "Employee added successfully",
                "data": data
            }

        # For complete change of that particular record
        @app.put("/emp")
        def update_emp(id: int, emp1: dict):
            for index, employee in enumerate(emp):
                if employee.get("id") == id:
                    emp[index] = emp1
                    return {
                        "message": "Employee updated successfully",
                        "data": emp1
                    }

        @app.delete("/emp")
        def delete_emp(id: int):
            for index, employee in enumerate(emp):
                if employee.get("id") == id:
                    deleted_emp = emp.pop(index)
                    return {
                        "message": "Employee deleted successfully",
                        "data": deleted_emp
                    }

        @app.get("/emp/{id}")
        def get_employee(id: int):
            for employee in emp:
                if employee.get("id") == id:
                    return employee

    [PAR]
        HTTP Request Parameters

            1. Path Parameter
            Syntax:
            https://<link>/<route-name>/{parameter}

            Example:
            https://api.example.com/users/101

            Explanation:
            Used to identify a specific resource (e.g., user with ID 101).

            from fastapi import FastAPI

            app = FastAPI()

            @app.get("/users/{user_id}")
            def get_user(user_id: int):
                return {"user_id": user_id}

            Request:
            GET /users/101


            2. Query Parameter
            Syntax:
            https://<link>/<route-name>?<key>=<value>

            Example:
            https://api.example.com/users?active=true

            Explanation:
            Used to filter, search, sort, or customize the returned results (e.g., only active users).

            from fastapi import FastAPI

            app = FastAPI()

            @app.get("/users")
            def get_users(active: bool):
                return {"active": active}

            Request:
            GET /users?active=true

            Syntax:
            POST/PUT/PATCH https://<link>/<route-name>

            Request Body:
            {
                "name": "Alice",
                "age": 25
            }

            Explanation:
            Used to send structured data in the request body to create or update a resource.

Response Model (FastAPI)

    Definition:
    A Pydantic model that defines and validates the structure of the API response.

    Syntax:
    @app.get("/users/{id}", response_model=UserResponse)

    Example:
    Request:
    GET /users/1

    Returned by API:
    {
        "id": 1,
        "name": "Alice",
        "password": "secret"
    }

    Response Sent:
    {
        "id": 1,
        "name": "Alice"
    }

    Note:
    Fields not defined in the response_model (e.g., "password") are automatically excluded.

Async Client (FastAPI)

    Definition:
    An async client is used to make non-blocking HTTP requests, allowing multiple tasks to run concurrently.

    Example:
    import httpx

    async with httpx.AsyncClient() as client:
        response = await client.get("https://api.example.com/users")

    Note:
    AsyncClient improves performance for I/O operations by waiting without blocking the application.

Generic (Pydantic) – Short Notes

    Definition:

    * Generic allows you to create reusable models that work with different data types.
    * It uses Python's typing.Generic with TypeVar.

    Import:

    from typing import Generic, TypeVar
    from pydantic import BaseModel

    T = TypeVar("T")

    Example:

    [USA]
        from typing import Generic, TypeVar
        from pydantic import BaseModel

        T = TypeVar("T")

        class Response(BaseModel, Generic[T]):
            data: T
            success: bool

        user = Response[str](data="Alice", success=True)
        number = Response[int](data=100, success=True)

    [RES]:

        Response[str](data="Alice", success=True)
        Response[int](data=100, success=True)

    Key Points:

    * T = TypeVar("T") → Defines a placeholder type.
    * Generic[T] → Makes the class generic.
    * Response[str] → data must be str.
    * Response[int] → data must be int.

    Memory Tip:
    Generic = One model, many data types.


Field Validator (Pydantic)

    [WHY]
        -> Validate or modify a single field before or after validation.

    [LIB]
        from pydantic import BaseModel, field_validator

    [SYN]
        class Model(BaseModel):
            field: type

            @field_validator("field")
            @classmethod
            def validate_field(cls, value):
                return value

    [PAR]
        "field" -> Field name to validate.
        value   -> Current field value.
        cls     -> Class reference.

    [USA]
        from pydantic import BaseModel, field_validator

        class User(BaseModel):
            age: int

            @field_validator("age")
            @classmethod
            def check_age(cls, value):
                if value < 18:
                    raise ValueError("Age must be at least 18")
                return value

        user = User(age=20)

    [RES]
        User(age=20)

    [ANE]
        from pydantic import BaseModel, field_validator

        class User(BaseModel):
            name: str

            @field_validator("name")
            @classmethod
            def capitalize_name(cls, value):
                return value.title()

        user = User(name="john doe")

    [RES]
        User(name='John Doe')

    [NOT]
        -> Must return the validated value.
        -> Raise ValueError if validation fails.
        -> Use one validator for one or more fields.

    Memory Tip:
        field_validator = Validate or transform one field.

Authentication (HTTP Basic)

    [WHY]
        -> Protect API endpoints using username and password.
        -> Allow only authenticated users to access resources.

    [LIB]
        from fastapi import FastAPI, Depends, HTTPException, status
        from fastapi.security import HTTPBasic, HTTPBasicCredentials
        import secrets

    [SYN]
        security = HTTPBasic()

        def authenticate(
            credentials: HTTPBasicCredentials = Depends(security)
        ):
            ...

    [PAR]
        HTTPBasic()
            -> Enables HTTP Basic Authentication.

        HTTPBasicCredentials
            -> Contains username and password.

        Depends()
            -> Injects authentication dependency.

        secrets.compare_digest()
            -> Securely compares credentials.

    [USA]
        from fastapi import FastAPI, Depends, HTTPException, status
        from fastapi.security import HTTPBasic, HTTPBasicCredentials
        import secrets

        app = FastAPI()

        security = HTTPBasic()

        def authenticate(
            credentials: HTTPBasicCredentials = Depends(security)
        ):
            username = secrets.compare_digest(
                credentials.username,
                "admin"
            )

            password = secrets.compare_digest(
                credentials.password,
                "admin123"
            )

            if not (username and password):
                raise HTTPException(
                    status_code=status.HTTP_401_UNAUTHORIZED,
                    detail="Invalid username or password"
                )

            return credentials.username

        @app.get("/profile")
        def profile(user: str = Depends(authenticate)):
            return {"user": user}

    [RES]
        Valid Login:
            {
                "user": "admin"
            }

        Invalid Login:
            Status Code: 401

            {
                "detail": "Invalid username or password"
            }

    [NOT]
        -> Use Depends() to protect routes.
        -> Use secrets.compare_digest() for secure comparison.
        -> HTTP Basic sends credentials with every request.
        -> Best used with HTTPS.

    Memory Tip:
        HTTPBasic + Depends = Secure endpoint with username & password.

                raise HTTPException(
                    status_code=403,
                    detail="Invalid API Key"
                )

            return key

        @app.get("/data")
        def get_data(
            api_key: str = Depends(verify_api_key)
        ):
            return {
                "message": "Access granted"
            }

OAuth2 + JWT Authentication (FastAPI)

    [WHY]
        -> Authenticate users using access tokens instead of sharing passwords.
        -> Allows secure communication between client, authorization server, and resource server.
        -> Token contains user identity and permissions.

    [LIB]
        [INS]
            pip install python-jose[crypto]
            pip install passlib[bcrypt]
            pip install python-multipart

        [USA]
            from datetime import datetime, timedelta

            from fastapi import (
                Depends,
                FastAPI,
                HTTPException,
                status
            )

            from fastapi.security import (
                OAuth2PasswordBearer,
                OAuth2PasswordRequestForm
            )

            from jose import JWTError, jwt

            from passlib.context import CryptContext


    [PAR]
        OAuth2PasswordBearer
            -> Reads Bearer token from Authorization header.

        OAuth2PasswordRequestForm
            -> Reads username and password from login form.

        JWT
            -> Encoded token containing user information.

        SECRET_KEY
            -> Used to sign and verify JWT token.

        CryptContext
            -> Hashes and verifies passwords.


    [FLOW]

        User
        |
        | username + password
        v
        Authorization Server
        |
        | generates JWT token
        v
        Access Token
        |
        | Authorization: Bearer token
        v
        Resource Server
        |
        v
        Protected Data


    [USA]

        from datetime import datetime, timedelta

        from fastapi import (
            Depends,
            FastAPI,
            HTTPException,
            status
        )

        from fastapi.security import (
            OAuth2PasswordBearer,
            OAuth2PasswordRequestForm
        )

        from jose import JWTError, jwt

        from passlib.context import CryptContext


        app = FastAPI()


        SECRET_KEY = "mysecretkey"

        ALGORITHM = "HS256"

        ACCESS_TOKEN_EXPIRE_MINUTES = 30


        pwd_context = CryptContext(
            schemes=["bcrypt"],
            deprecated="auto"
        )


        oauth2_scheme = OAuth2PasswordBearer(
            tokenUrl="login"
        )


        users = {
            "admin": {
                "username": "admin",
                "password": pwd_context.hash("admin123")
            }
        }


        def verify_password(
            plain_password,
            hashed_password
        ):
            return pwd_context.verify(
                plain_password,
                hashed_password
            )


        def create_access_token(data: dict):

            expire = datetime.utcnow() + timedelta(
                minutes=ACCESS_TOKEN_EXPIRE_MINUTES
            )

            data.update(
                {
                    "exp": expire
                }
            )

            return jwt.encode(
                data,
                SECRET_KEY,
                algorithm=ALGORITHM
            )


        @app.post("/login")
    def login(
            form_data: OAuth2PasswordRequestForm = Depends()
        ):

            user = users.get(
                form_data.username
            )

            if not user:
                raise HTTPException(
                    status_code=401,
                    detail="Invalid user"
                )

            if not verify_password(
                form_data.password,
                user["password"]
            ):
                raise HTTPException(
                    status_code=401,
                    detail="Invalid password"
                )

            token = create_access_token(
                {
                    "sub": user["username"]
                }
            )

            return {
                "access_token": token,
                "token_type": "bearer"
            }


        @app.get("/profile")
    def profile(
            token: str = Depends(oauth2_scheme)
        ):

            try:
                payload = jwt.decode(
                    token,
                    SECRET_KEY,
                    algorithms=[ALGORITHM]
                )

                username = payload.get("sub")

                return {
                    "user": username
                }

            except JWTError:
                raise HTTPException(
                    status_code=401,
                    detail="Invalid token"
                )


    [RES]

        Login Request:

            username=admin
            password=admin123


        Response:

            {
                "access_token": "eyJhbGciOiJIUzI1...",
                "token_type": "bearer"
            }


        API Request:

            Authorization:
            Bearer eyJhbGciOiJIUzI1...


        Response:

            {
                "user": "admin"
            }


    [NOT]
        -> OAuth2 does not freeze or share passwords.
        -> It shares access tokens with limited permissions.
        -> Resource Server stores protected resources, not credentials.
        -> Authorization Server manages authentication and token creation.
        -> JWT tokens should have expiry time.
        -> Always use HTTPS.


    [USC]
        -> Login systems.
        -> Single Sign-On (SSO).
        -> Mobile apps.
        -> Third-party API authorization.
        -> Microservice authentication.


    Memory Tip:

        OAuth2 = Login → Generate Token → Send Token → Access Resource.

            db = SessionLocal()

            try:
                yield db

            finally:
                db.close()


    [RES]

        SQLAlchemy database session object


    [NOT]

        -> engine = Database connection.
        -> SessionLocal = Creates database sessions.
        -> Base = Parent class for database models.
        -> get_db() = Provides and closes database connection.

        from sqlalchemy.orm import Session

        from database import get_db
        from models import Student

    [USA]

        from fastapi import APIRouter, Depends
        from sqlalchemy.orm import Session

        from database import get_db
        from models import Student
        from schemas import StudentCreate


        router = APIRouter()


        @router.post("/students")
        def create_student(
            student: StudentCreate,
            db: Session = Depends(get_db)
        ):

            new_student = Student(
                name=student.name,
                age=student.age,
                city=student.city
            )


            db.add(new_student)

            db.commit()

            db.refresh(new_student)


            return new_student



        @router.get("/students")
        def get_students(
            db: Session = Depends(get_db)
        ):

            students = db.query(
                Student
            ).all()

            return students

        db.add()
            -> Insert new record.

            -> Save changes permanently.

        db.refresh()
            -> Reload database object.

        db.query()
