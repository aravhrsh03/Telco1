class CallUsageAnalyzer:

class Outage:
    def __init__(self, area_code, duration_minutes, severity):
        self.area_code = area_code
        self.duration_minutes = duration_minutes
        self.severity = severity

class BroadbandOutageTracker:
    VALID_SEVERITIES = {"minor", "major", "critical"}

    def __init__(self):
        self.outage_records = []

    def add_outage_record(self, area_code, duration_minutes, severity):
        if area_code is None or area_code.strip() == "":
            raise ValueError("area_code is empty")
        if duration_minutes <= 0:
            raise ValueError("duration_minutes cannot be negative")
        if severity.lower() not in self.VALID_SEVERITIES:
            raise ValueError("severity not found")
        
        self.outage_records.append(Outage(area_code, duration_minutes, severity))

    def total_downtime_per_area(self):
        total_downtime_by_area = dict()

        for outage_log in self.outage_records:
            current_area = outage_log.area_code
            current_duration = outage_log.duration_minutes

            if current_area not in self.outage_records:
                total_downtime_by_area[current_area] = current_duration
            else:
                total_downtime_by_area[current_area] += current_duration
        
        return total_downtime_by_area

    def outage_count_per_severity(self):
        outage_count_by_severity = {
            self.VALID_SEVERITIES[0]: 0,
            self.VALID_SEVERITIES[1]: 0,
            self.VALID_SEVERITIES[2]: 0
        }

        for outage_log in self.outage_records:
            outage_count_by_severity[outage_log.severity] += 1
        
        return outage_count_by_severity

    def most_affected_area(self):
        if self.outage_records == []:
            return None

        affected_area_with_downtime = self.total_downtime_per_area()

        area_code_ = None
        duration_ = 0

        for area_code_key, duration_value in affected_area_with_downtime.items():
            if duration_ < duration_value:
                duration_ = duration_value
                area_code = area_code_key
            
            elif duration_ == duration_value:
                if area_code_ > area_code_key:
                    area_code_ = area_code_key
        
        return area_code_

    def total_outages(self):
        return len(self.outage_records)
